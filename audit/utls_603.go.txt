type record struct {
 IP string `json:"ip"`
 Mode string `json:"mode"`
 Status int `json:"status,omitempty"`
 TLSVerified bool `json:"tls_verified"`
 FinalURL string `json:"final_url,omitempty"`
 Redirects []string `json:"redirects,omitempty"`
 Bytes int `json:"bytes,omitempty"`
 BodySHA256 string `json:"body_sha256,omitempty"`
 Signal string `json:"signal,omitempty"`
 Error string `json:"error,omitempty"`
}
type result struct { Host string `json:"host"`; Addresses []string `json:"addresses"`; Attempts []record `json:"attempts"` }
func public(ip net.IP) bool {
 return ip.IsGlobalUnicast() && !ip.IsPrivate() && !ip.IsLoopback() && !ip.IsLinkLocalUnicast() && !ip.IsUnspecified() && !ip.IsMulticast()
}
func resolve(ctx context.Context, host string)([]net.IP,error){
 a,e:=net.DefaultResolver.LookupIPAddr(ctx,host);if e!=nil{return nil,e}
 var v4,v6 []net.IP
 for _,x:=range a {if !public(x.IP){continue};if x.IP.To4()!=nil {if len(v4)<2{v4=append(v4,x.IP)}} else if len(v6)<1{v6=append(v6,x.IP)}}
 return append(v4,v6...),nil
}
func main(){
 if len(os.Args)!=2 {os.Exit(2)}
 host:=strings.ToLower(strings.TrimSuffix(os.Args[1],"."))
 ctx,cancel:=context.WithTimeout(context.Background(),40*time.Second);defer cancel()
 ips,err:=resolve(ctx,host);out:=result{Host:host}
 if err!=nil {out.Attempts=append(out.Attempts,record{Mode:"dns",Error:err.Error()});json.NewEncoder(os.Stdout).Encode(out);return}
 for _,ip:=range ips {out.Addresses=append(out.Addresses,ip.String())}
 for _,ip:=range ips{
  for _,mode:=range []string{"utls-chrome-h2","utls-chrome-http1","http1-plain"} {
   if ctx.Err()!=nil {break}
   rec:=record{IP:ip.String(),Mode:mode}
   dial:=func(ctx context.Context,network,addr string)(net.Conn,error){
    dest,port,e:=net.SplitHostPort(addr);if e!=nil{return nil,e}
    chosen:=ip
    if !strings.EqualFold(dest,host){resolved,e:=resolve(ctx,dest);if e!=nil{return nil,e};if len(resolved)==0{return nil,fmt.Errorf("redirect has no public IP")};chosen=resolved[0]}
    if !public(chosen){return nil,fmt.Errorf("non-public IP refused")}
    return (&net.Dialer{Timeout:4*time.Second}).DialContext(ctx,"tcp",net.JoinHostPort(chosen.String(),port))
   }
   tlsDial:=func(ctx context.Context,network,addr string)(net.Conn,error){
    server,_,e:=net.SplitHostPort(addr);if e!=nil{return nil,e}
    c,e:=dial(ctx,network,addr);if e!=nil{return nil,e}
    u:=utls.UClient(c,&utls.Config{ServerName:server},utls.HelloCustom)
    spec,e:=utls.UTLSIdToSpec(utls.HelloChrome_Auto);if e!=nil{c.Close();return nil,e}
    alpn:="http/1.1";if mode=="utls-chrome-h2"{alpn="h2"}
    for _,ext:=range spec.Extensions {if a,ok:=ext.(*utls.ALPNExtension);ok{a.AlpnProtocols=[]string{alpn}}}
    if e=u.ApplyPreset(&spec);e!=nil{c.Close();return nil,e}
    if e=u.HandshakeContext(ctx);e!=nil{c.Close();return nil,e}
    if mode=="utls-chrome-h2" && u.ConnectionState().NegotiatedProtocol!="h2" {c.Close();return nil,fmt.Errorf("h2 not negotiated")}
    return u,nil
   }
   var rt http.RoundTripper
   if mode=="utls-chrome-h2"{
    rt=&http2.Transport{DialTLSContext:func(ctx context.Context,network,addr string,_ *tls.Config)(net.Conn,error){return tlsDial(ctx,network,addr)}}
   }else{
    rt=&http.Transport{Proxy:nil,DisableKeepAlives:true,ForceAttemptHTTP2:false,DialContext:dial,
      DialTLSContext:func(ctx context.Context,network,addr string)(net.Conn,error){return tlsDial(ctx,network,addr)}}
   }
   redirects:=[]string{}
   client:=http.Client{Transport:rt,Timeout:5*time.Second,CheckRedirect:func(req *http.Request,via []*http.Request)error{
    if len(via)>=5{return fmt.Errorf("redirect limit")}
    if req.URL.User!=nil || req.URL.Hostname()=="" || (req.URL.Scheme!="http" && req.URL.Scheme!="https"){return fmt.Errorf("unsafe redirect")}
    if port:=req.URL.Port();port!="" && port!="80" && port!="443"{return fmt.Errorf("nonstandard redirect port")}
    if mode=="utls-chrome-h2" && req.URL.Scheme!="https"{return fmt.Errorf("h2 redirect to http")}
    req.URL.RawQuery="";req.URL.Fragment=""
    redirects=append(redirects,req.URL.String());return nil
   }}
   scheme:="https";if mode=="http1-plain"{scheme="http"}
   req,_:=http.NewRequestWithContext(ctx,"GET",scheme+"://"+host+"/",nil)
   req.Header.Set("User-Agent","Geosite-RU-Audit/1.0 (+https://github.com/SpeedeMiner/Geosite-ru-light)")
   resp,e:=client.Do(req)
   if e!=nil {rec.Error=e.Error()}else{
    b,readErr:=io.ReadAll(io.LimitReader(resp.Body,16384));resp.Body.Close()
    rec.Status=resp.StatusCode;rec.TLSVerified=resp.Request.URL.Scheme=="https";rec.FinalURL=resp.Request.URL.String();rec.Redirects=redirects;rec.Bytes=len(b)
    hash:=sha256.Sum256(b);rec.BodySHA256=hex.EncodeToString(hash[:])
    low:=strings.ToLower(string(b))
    for _,marker:=range []string{"domain is for sale","domain for sale","домен продается","домен припаркован","доменное имя продается"} {if strings.Contains(low,marker){rec.Signal="parking";break}}
    if rec.Signal=="" {for _,marker:=range []string{"captcha","checking your browser","cloudflare ray id"} {if strings.Contains(low,marker){rec.Signal="waf_or_captcha";break}}}
    if readErr!=nil {rec.Error=readErr.Error()}
   }
   // Prevent queries or fragments from entering the evidence.
   if u,e:=url.Parse(rec.FinalURL);e==nil && u!=nil {u.RawQuery="";u.Fragment="";rec.FinalURL=u.String()}
   out.Attempts=append(out.Attempts,rec)
  }
 }
 json.NewEncoder(os.Stdout).Encode(out)
}
